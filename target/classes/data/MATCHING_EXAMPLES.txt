function calculateMatchScore(project, candidate) {
  let score = 0;
  let totalSkills = project.requiredSkills.length;
  
  for (let req of project.requiredSkills) {
    let candidateSkill = findSkill(candidate, req.skillName);
    
    if (candidateSkill) {
      score += 10; // base skill match
      
      if (levelValue(candidateSkill.level) >= levelValue(req.level)) {
        score += 5; // bonus for meeting level requirement
      } else if (levelValue(candidateSkill.level) >= levelValue(req.level) - 1) {
        score += 3; // partial bonus if one level below
      }
    } else {
      score -= 2; // penalty for missing skill
    }
  }
  
  // Normalize to 0-100
  return Math.max(0, (score / (totalSkills * 15)) * 100);
}

function levelValue(level) {
  return { "Junior": 1, "Mid": 2, "Senior": 3 }[level] || 0;
}
